# Pip-only installation of the reproduction dependencies.
#
# Role: this file installs into an existing Python 3.10 interpreter.
# environment.yaml is the canonical specification for the A-board code
# inspection and additionally pins the interpreter itself. The two files pin
# IDENTICAL versions; change both together or neither.
#
# JittorGeometric is not on PyPI. Install it after this file:
#   pip install --no-build-isolation git+https://github.com/AlgRUC/JittorGeometric.git

# --- numerical core -------------------------------------------------------
# numpy is held below 2.0: the frozen postprocessors depend on exact float64
# comparison and stable argsort semantics, and the accepted member hashes were
# produced under the 1.x behaviour.
numpy==1.26.4
pandas==2.2.3
scipy==1.15.2
scikit-learn==1.6.1

# --- gradient-boosted ranker (dataset1 and dataset2 base score matrices) ---
lightgbm==4.5.0

# --- progress reporting in the embedding trainers -------------------------
tqdm==4.67.1

# --- mandatory competition framework --------------------------------------
# Used by src/train_line_jt.py and src/train_bpr_jt.py. Compiles its own kernels
# on first import and needs g++ >= 5.4; the GPU path additionally needs a CUDA
# toolkit. See docs/competition/ab_algorithm_consistency_contract.md.
jittor==1.3.10

# --- REQUIRED: shared utilities on the ranking path ------------------------
# torch is NOT optional. src/train_line.py is both the historical PyTorch
# trainer AND the shared utility module (PROJECT_ROOT, rownorm, build_sim_cache,
# train_csv, c_cols) imported by src/ensemble_predict.py, src/ranker_ds1.py,
# src/ranker_basket_ds2.py and src/ds2_mf_basket_pack.py. Without torch those
# four stages fail at import with ModuleNotFoundError, so no ranking stage runs
# and no submission member can be produced -- regardless of which trainer
# produced the embeddings.
#
# This does not weaken the framework requirement: the Jittor trainers remain the
# compliant model path. On that path torch supplies utility functions only and a
# CPU build is sufficient. A CUDA build is needed only to re-run the historical
# PyTorch trainers.
#
# Left unpinned deliberately: no accepted artifact hash depends on the torch
# version, and pinning a CUDA build would force a multi-gigabyte download on
# reviewers who only need the utilities. Install the CPU wheel to keep it small:
#   pip install torch --index-url https://download.pytorch.org/whl/cpu
torch
